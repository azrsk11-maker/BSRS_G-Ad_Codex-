    /** Purpose-specific local observation; the enclosing journal retains lease ownership. */
    public function localHistoryCheckpoint(\BSRE\ClosedRetention\LeaseBoundWpdb $db, RuntimePolicy $policy, CanonicalCaptureProducer $producer): array
    {
        if ($db!==$this->db || $policy!==$this->policy || $producer!==$this->receiverProducer) RuntimePolicy::fail('G39_JOURNAL_COMPOSITION_CONFLICT');
        $this->assertCurrentLease();
        if ($this->scalar('SELECT @@session.in_transaction')!=='1' || $this->scalar('SELECT @@session.tx_read_only')!=='1'
            || $this->scalar('SELECT @@session.tx_isolation')!=='REPEATABLE-READ') RuntimePolicy::fail('G39_READ_ONLY_SNAPSHOT_REQUIRED');
        $s=$this->state(); $row=$this->raw(); $anchor=$this->custody->current();
        if ($row===null || $anchor!==['revision'=>(int)$row['revision'],'hash'=>$row['body_hash']]) RuntimePolicy::fail('G5_CURRENT_ANCHOR_MISMATCH');
        return ['scope'=>$policy->scope(),'anchor'=>$anchor,'cipher_sha256'=>$row['body_hash'],
            'incomplete'=>$s['incomplete'],'admission_floor'=>$s['admission_floor']];
    }
